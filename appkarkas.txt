# ════════════════════════════════════════════════════════
#  DATABASE
# ════════════════════════════════════════════════════════
class DB:
    conn = None

    @classmethod
    def connect(cls):
        cls.conn = psycopg2.connect(
            host=DB_HOST, port=DB_PORT, dbname=DB_NAME,
            user=DB_USER, password=DB_PASSWORD
        )
        cls.conn.autocommit = False

    @classmethod
    def q(cls, sql, params=None, fetch=None):
        with cls.conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
            cur.execute(sql, params)
            if fetch == "one": return cur.fetchone()
            if fetch == "all": return cur.fetchall()
            cls.conn.commit()

    @classmethod
    def columns(cls, table):
        return cls.q(
            "SELECT column_name, data_type FROM information_schema.columns "
            "WHERE table_name=%s AND table_schema='public' ORDER BY ordinal_position",
            (table,), fetch="all"
        ) or []

    @classmethod
    def tables(cls):
        rows = cls.q(
            "SELECT table_name FROM information_schema.tables "
            "WHERE table_schema='public' AND table_type='BASE TABLE' ORDER BY 1",
            fetch="all"
        )
        return [r["table_name"] for r in rows] if rows else []

    @classmethod
    def hash(cls, pw): return hashlib.sha256(pw.encode()).hexdigest()

    @classmethod
    def login(cls, login, pw):
        return cls.q(
            "SELECT u.id, u.login, u.full_name, r.role_name "
            "FROM users u JOIN roles r ON u.role_id=r.id "
            "WHERE u.login=%s AND u.password_hash=%s",
            (login, cls.hash(pw)), fetch="one"
        )

    @classmethod
    def register(cls, login, pw, full_name, phone=""):
        role = cls.q("SELECT id FROM roles WHERE role_name=%s", (ROLE_USER,), fetch="one")
        if not role:
            raise ValueError(f"Роль '{ROLE_USER}' не найдена")
        cls.q(
            "INSERT INTO users (login,password_hash,role_id,full_name,phone) "
            "VALUES (%s,%s,%s,%s,%s)",
            (login, cls.hash(pw), role["id"], full_name, phone)
        )

                self._status(self.lbl_err, "✓ Вход выполнен", ok=True)

                self._status(self.lbl_err, "✗ Неверный логин или пароль")

            self._status(self.lbl_err, f"✗ {e}")

            self._status(self.lbl_reg, "✗ Пароли не совпадают"); return
        if len(pw) < 6:
            self._status(self.lbl_reg, "✗ Пароль слишком короткий"); return

            self._status(self.lbl_reg, "✓ Готово! Войдите.", ok=True)
            self.e_login.setText(login)
            self.tabs.setCurrentIndex(0)
        except psycopg2.errors.UniqueViolation:
            DB.conn.rollback()
            self._status(self.lbl_reg, "✗ Логин уже занят")
        except Exception as e:
            try: DB.conn.rollback()
            except: pass
            self._status(self.lbl_reg, f"✗ {e}")

class RowDialog(QDialog):
    def __init__(self, table, columns, existing=None, parent=None):
        super().__init__(parent)
        self.table = table; self.columns = columns; self.existing = existing
        self.setWindowTitle("Редактировать" if existing else "Добавить")
        self.setMinimumWidth(460); self.setStyleSheet(STYLE)
        self.inputs = {}; self._build()

    def _build(self):
        lay = QVBoxLayout(self); lay.setSpacing(10); lay.setContentsMargins(22,22,22,22)
        t = QLabel("Редактировать запись" if self.existing else "Новая запись")
        t.setObjectName("title"); lay.addWidget(t)

        scroll = QScrollArea(); scroll.setWidgetResizable(True)
        scroll.setFrameShape(QFrame.Shape.NoFrame)
        inner = QWidget(); form = QFormLayout(inner); form.setSpacing(9)
        scroll.setWidget(inner); lay.addWidget(scroll)

        auto = {"id", "created_at"} if not self.existing else set()
        for col in self.columns:
            name = col["column_name"]; dtype = col["data_type"]
            if name in auto: continue
            lbl = QLabel(name); lbl.setObjectName("dim")
            if self.existing:
                w = QLineEdit(str(self.existing.get(name) or ""))
                if name in ("id","created_at"):
                    w.setReadOnly(True)
                    w.setStyleSheet("color:#555;background:#16213e;")
            elif "integer" in dtype or "serial" in dtype:
                w = QSpinBox(); w.setRange(0, 9_999_999)
            elif "numeric" in dtype or "decimal" in dtype:
                w = QDoubleSpinBox(); w.setRange(0, 9_999_999); w.setDecimals(2)
            elif "text" in dtype:
                w = QTextEdit(); w.setFixedHeight(70)
            else:
                w = QLineEdit()
            self.inputs[name] = w; form.addRow(lbl, w)

        btns = QHBoxLayout()
        c = QPushButton("Отмена"); c.setObjectName("secondary"); c.clicked.connect(self.reject)
        s = QPushButton("Сохранить"); s.setObjectName("primary"); s.clicked.connect(self._save)
        btns.addStretch(); btns.addWidget(c); btns.addWidget(s); lay.addLayout(btns)

    def _val(self, name):
        w = self.inputs.get(name)
        if w is None: return None
        if isinstance(w, QTextEdit): return w.toPlainText() or None
        if isinstance(w, QDoubleSpinBox): return w.value()
        if isinstance(w, QSpinBox): return w.value()
        v = w.text().strip(); return v or None

    def _save(self):
        try:
            auto = {"id","created_at"}
            if self.existing:
                cols = [c["column_name"] for c in self.columns if c["column_name"] not in auto]
                sets = ", ".join(f"{c}=%s" for c in cols)
                vals = [self._val(c) for c in cols]
                pk = "id" if "id" in self.existing else list(self.existing)[0]
                DB.q(f"UPDATE {self.table} SET {sets} WHERE {pk}=%s", vals+[self.existing[pk]])
            else:
                cols = [c["column_name"] for c in self.columns if c["column_name"] not in auto]
                DB.q(f"INSERT INTO {self.table} ({','.join(cols)}) VALUES ({','.join(['%s']*len(cols))})",
                     [self._val(c) for c in cols])
            self.accept()
        except Exception as e:
            try: DB.conn.rollback()
            except: pass
            QMessageBox.critical(self, "Ошибка", str(e))

class TableViewer(QWidget):
    """Только чтение — без кнопок редактирования."""
    def __init__(self, table):
        super().__init__()
        self.table = table; self.columns = []; self.data = []
        self._build(); self.refresh()

    def _build(self):
        lay = QVBoxLayout(self); lay.setContentsMargins(0,0,0,0); lay.setSpacing(7)
        tb = QHBoxLayout()
        self.search = QLineEdit(); self.search.setPlaceholderText("🔍 Поиск...")
        self.search.setFixedHeight(34); self.search.textChanged.connect(self._filter)
        tb.addWidget(self.search, stretch=1)
        ref = QPushButton("↻ Обновить"); ref.setObjectName("secondary")
        ref.clicked.connect(self.refresh); tb.addWidget(ref)
        lay.addLayout(tb)

        self.tbl = QTableWidget()
        self.tbl.setAlternatingRowColors(True)
        self.tbl.setSelectionBehavior(QTableWidget.SelectionBehavior.SelectRows)
        self.tbl.setEditTriggers(QTableWidget.EditTrigger.NoEditTriggers)
        self.tbl.horizontalHeader().setStretchLastSection(True)
        self.tbl.verticalHeader().setVisible(False)
        lay.addWidget(self.tbl)

        self.status = QLabel(""); self.status.setObjectName("dim")
        lay.addWidget(self.status)

    def refresh(self):
        try:
            self.columns = DB.columns(self.table)
            self.data = DB.q(f"SELECT * FROM {self.table} ORDER BY 1", fetch="all") or []
            self._fill(self.data)
            self.status.setText(f"  {len(self.data)} записей  |  только просмотр")
        except Exception as e:
            self.status.setText(f"  Ошибка: {e}")

    def _fill(self, rows):
        names = [c["column_name"] for c in self.columns]
        self.tbl.setColumnCount(len(names))
        self.tbl.setHorizontalHeaderLabels(names)
        self.tbl.setRowCount(len(rows))
        for r, row in enumerate(rows):
            for c, n in enumerate(names):
                v = row.get(n)
                self.tbl.setItem(r, c, QTableWidgetItem("" if v is None else str(v)))
        self.tbl.resizeColumnsToContents()

    def _filter(self, text):
        t = text.lower()
        rows = [r for r in self.data if any(t in str(v).lower() for v in r.values())] if t else self.data
        self._fill(rows)
        self.status.setText(
            f"  {len(rows)} из {len(self.data)}  |  только просмотр" if t
            else f"  {len(self.data)} записей  |  только просмотр"
        )

class TableEditor(QWidget):
    def __init__(self, table):
        super().__init__()
        self.table = table; self.columns = []; self.data = []
        self._build(); self.refresh()

    def _build(self):
        lay = QVBoxLayout(self); lay.setContentsMargins(0,0,0,0); lay.setSpacing(7)
        tb = QHBoxLayout()
        self.search = QLineEdit(); self.search.setPlaceholderText("🔍 Поиск...")
        self.search.setFixedHeight(34); self.search.textChanged.connect(self._filter)
        tb.addWidget(self.search, stretch=1)
        for txt, obj, slot in [("↻","secondary",self.refresh),
                                ("+  Добавить","primary",self._add),
                                ("✏  Изменить","secondary",self._edit),
                                ("✕  Удалить","danger",self._delete)]:
            b = QPushButton(txt); b.setObjectName(obj); b.clicked.connect(slot)
            tb.addWidget(b)
        lay.addLayout(tb)

        self.tbl = QTableWidget()
        self.tbl.setAlternatingRowColors(True)
        self.tbl.setSelectionBehavior(QTableWidget.SelectionBehavior.SelectRows)
        self.tbl.setEditTriggers(QTableWidget.EditTrigger.NoEditTriggers)
        self.tbl.horizontalHeader().setStretchLastSection(True)
        self.tbl.verticalHeader().setVisible(False)
        self.tbl.doubleClicked.connect(self._edit)
        lay.addWidget(self.tbl)

        self.status = QLabel(""); self.status.setObjectName("dim")
        lay.addWidget(self.status)

    def refresh(self):
        try:
            self.columns = DB.columns(self.table)
            self.data = DB.q(f"SELECT * FROM {self.table} ORDER BY 1", fetch="all") or []
            self._fill(self.data); self.status.setText(f"  {len(self.data)} записей")
        except Exception as e:
            self.status.setText(f"  Ошибка: {e}")

    def _fill(self, rows):
        names = [c["column_name"] for c in self.columns]
        self.tbl.setColumnCount(len(names))
        self.tbl.setHorizontalHeaderLabels(names)
        self.tbl.setRowCount(len(rows))
        for r, row in enumerate(rows):
            for c, n in enumerate(names):
                v = row.get(n)
                self.tbl.setItem(r, c, QTableWidgetItem("" if v is None else str(v)))
        self.tbl.resizeColumnsToContents()

    def _filter(self, text):
        t = text.lower()
        rows = [r for r in self.data if any(t in str(v).lower() for v in r.values())] if t else self.data
        self._fill(rows)
        self.status.setText(f"  {len(rows)} из {len(self.data)}" if t else f"  {len(self.data)} записей")

    def _sel(self):
        row = self.tbl.currentRow()
        if row < 0: return None
        names = [c["column_name"] for c in self.columns]
        return {names[i]: self.tbl.item(row, i).text() for i in range(len(names))}

    def _add(self):
        if RowDialog(self.table, self.columns, parent=self).exec(): self.refresh()

    def _edit(self):
        s = self._sel()
        if not s: QMessageBox.information(self,"","Выберите запись"); return
        if RowDialog(self.table, self.columns, existing=s, parent=self).exec(): self.refresh()

    def _delete(self):
        s = self._sel()
        if not s: QMessageBox.information(self,"","Выберите запись"); return
        pk = "id" if "id" in s else list(s)[0]
        if QMessageBox.question(self,"Удалить",f"Удалить {pk}={s[pk]}?",
            QMessageBox.StandardButton.Yes|QMessageBox.StandardButton.No
        ) == QMessageBox.StandardButton.Yes:
            try: DB.q(f"DELETE FROM {self.table} WHERE {pk}=%s", (s[pk],)); self.refresh()
            except Exception as e:
                try: DB.conn.rollback()
                except: pass
                QMessageBox.critical(self,"Ошибка",str(e))

def main():
    app = QApplication(sys.argv)
    app.setStyleSheet(STYLE)

    try:
        DB.connect()
    except Exception as e:
        QMessageBox.critical(None, "Ошибка подключения",
            f"Не удалось подключиться к БД:\n\n{e}\n\n"
            "Проверьте параметры в начале файла app.py")
        sys.exit(1)

    auth = AuthWindow()

    def on_login(user):
        auth.hide()
        role = user.get("role_name", "")

        if role == ROLE_ADMIN:
            win = AdminWindow(user)
        # GUEST_START
        elif role == ROLE_GUEST:
            win = GuestWindow(user)
        # GUEST_END
        else:
            win = UserWindow(user)

        win.logout.connect(lambda: auth.show())
        win.show()

    auth.logged_in.connect(on_login)
    auth.show()
    sys.exit(app.exec())


if __name__ == "__main__":
    main()
